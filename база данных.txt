@bot.callback_query_handler(func=lambda call:True)
def callback(call):
    conn = sqlite3.connect("studio_dvorik.sql")
    cur = conn.cursor()
    cur.execute("SELECT * FROM users")
    users = cur.fetchall()
    info = ""
    for el in users:
        info += f"Имя: {el[1]}, пароль: {el[2]}\n"
    cur.close()
    conn.close()

    bot.send_message(call.message.chat.id, info)
